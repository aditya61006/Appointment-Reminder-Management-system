
# 1. Create project
cd C:\xampp\htdocs
laravel new ProjectName

# 2. Enter project
cd ProjectName

# 3. Install dependencies
composer install
npm install

# 4. Generate application key
php artisan key:generate

# 5. Configure .env manually for MySQL
# Create database in phpMyAdmin first.

# 6. Clear configuration
php artisan config:clear

# 7. Run migrations
php artisan migrate

# 8. Start Laravel server
php artisan serve

# 9. Start frontend server in another terminal
npm run dev
